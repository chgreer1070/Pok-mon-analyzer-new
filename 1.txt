import cv2
import numpy as np

# -----------------------------------------------------------------------------
# --- USER TUNING AREA (CONFIG) ---
# -----------------------------------------------------------------------------

# --- 1. FILE & MODE ---
# IMPORTANT: Replace this with the filename of your high-resolution card scan
IMAGE_FILE = "YOUR_CARD_SCAN.JPG"

# Specify if you are scanning the 'front' or 'back' of the card.
# This is CRITICAL for the edge wear detection.
# Valid options: 'front' or 'back'
CARD_SIDE = 'front'

# --- 2. EDGE WEAR (WHITENING) TUNING ---
# These are HSV (Hue, Saturation, Value) ranges. You WILL need to tune
# these to match the specific lighting of your scanner.
# Use the 'defect_mask.png' output to help you tune.
# In the mask, "good" pixels are black, "defects" (whitening) are white.

# HSV range for the "good" yellow border on the FRONT
LOWER_YELLOW = np.array([20, 100, 100])
UPPER_YELLOW = np.array([30, 255, 255])

# HSV range for the "good" blue border on the BACK
LOWER_BLUE = np.array([90, 80, 50])
UPPER_BLUE = np.array([130, 255, 255])

# --- 3. CENTERING TUNING ---
# Adjust these Canny thresholds if the script fails to find the
# card's outer edge or the inner artwork box.
# Format: (lower_threshold, upper_threshold)
CENTERING_CANNY_THRESHOLDS = (50, 150)

# --- 4. SURFACE & PRINT LINE TUNING ---
# Canny thresholds for finding edges *within* the artwork box.
# These should be "tighter" (higher values) than the centering
# thresholds to avoid picking up normal artwork as "print lines".
SURFACE_CANNY_THRESHOLDS = (100, 200)

# Hough Line Transform parameters for print lines.
# - HOUGH_THRESHOLD: How many points are needed to "vote" for a line.
# - HOUGH_MIN_LINE_LENGTH: The shortest line (in pixels) to report.
#   (Set this to ~50% of your artwork box's height to find major lines)
# - HOUGH_MAX_LINE_GAP: Max gap (in pixels) between segments
#   to link them into one line.
HOUGH_THRESHOLD = 50
HOUGH_MIN_LINE_LENGTH = 150  # Tune this!
HOUGH_MAX_LINE_GAP = 20
# -----------------------------------------------------------------------------

def analyze_centering(img_gray, img_color, thresholds):
    """
    Finds the two largest contours (outer card and inner artwork box)
    and calculates centering.
    """
    print("Step 1: Analyzing Centering...")
    
    # Pre-process for contour detection
    blur = cv2.GaussianBlur(img_gray, (7, 7), 0)
    edges = cv2.Canny(blur, thresholds[0], thresholds[1])

    # Find all contours
    contours, _ = cv2.findContours(edges.copy(), cv2.RETR_TREE, cv2.CHAIN_APPROX_SIMPLE)

    if len(contours) < 2:
        print("  [ERROR] Could not find at least two contours.")
        print("  Try adjusting CENTERİNG_CANNY_THRESHOLDS or ensure")
        print("  the card is on a plain, contrasting background.")
        return None, None, None

    # Sort by area, largest to smallest
    sorted_contours = sorted(contours, key=cv2.contourArea, reverse=True)

    # Get the bounding boxes
    outer_box = cv2.boundingRect(sorted_contours[0])
    inner_box = cv2.boundingRect(sorted_contours[1])

    x_o, y_o, w_o, h_o = outer_box
    x_i, y_i, w_i, h_i = inner_box

    # Calculate border dimensions
    left_border = x_i - x_o
    right_border = (x_o + w_o) - (x_i + w_i)
    top_border = y_i - y_o
    bottom_border = (y_o + h_o) - (y_i + h_i)
    
    total_horizontal = left_border + right_border
    total_vertical = top_border + bottom_border

    if total_horizontal == 0 or total_vertical == 0:
        print("  [ERROR] Could not calculate centering. Borders have zero width.")
        return None, None, None

    # Calculate percentages
    left_percent = (left_border / total_horizontal) * 100
    right_percent = (right_border / total_horizontal) * 100
    top_percent = (top_border / total_vertical) * 100
    bottom_percent = (bottom_border / total_vertical) * 100

    # Draw boxes on the visualization image
    cv2.rectangle(img_color, (x_o, y_o), (x_o + w_o, y_o + h_o), (0, 255, 0), 3)
    cv2.rectangle(img_color, (x_i, y_i), (x_i + w_i, y_i + h_i), (0, 0, 255), 3)

    report = (
        f"  Horizontal (L/R): {left_percent:.2f}% / {right_percent:.2f}%\n"
        f"  Vertical (T/B):   {top_percent:.2f}% / {bottom_percent:.2f}%"
    )
    
    print("  Centering analysis complete.")
    return report, outer_box, inner_box

def analyze_edge_wear(img_color, outer_box, inner_box):
    """
    Analyzes the border area for pixels that are NOT the correct
    card border color (e.g., detects white or silver specks).
    """
    print("Step 2: Analyzing Edge Wear (Whitening)...")
    
    x_o, y_o, w_o, h_o = outer_box
    x_i, y_i, w_i, h_i = inner_box

    # Create a black image (mask) of the full card size
    outer_mask = np.zeros(img_color.shape[:2], dtype="uint8")
    cv2.rectangle(outer_mask, (x_o, y_o), (x_o + w_o, y_o + h_o), 255, -1)

    # Create a black mask of the inner artwork box
    inner_mask = np.zeros(img_color.shape[:2], dtype="uint8")
    cv2.rectangle(inner_mask, (x_i, y_i), (x_i + w_i, y_i + h_i), 255, -1)

    # The border mask is the outer mask MINUS the inner mask
    border_mask = cv2.subtract(outer_mask, inner_mask)

    # Get the total number of pixels in the border
    total_border_pixels = cv2.countNonZero(border_mask)
    if total_border_pixels == 0:
        print("  [ERROR] No border pixels found.")
        return "N/A"

    # Convert the whole image to HSV color space
    hsv_img = cv2.cvtColor(img_color, cv2.COLOR_BGR2HSV)

    # Select the correct color range based on the CARD_SIDE config
    if CARD_SIDE == 'front':
        lower_hsv = LOWER_YELLOW
        upper_hsv = UPPER_YELLOW
    elif CARD_SIDE == 'back':
        lower_hsv = LOWER_BLUE
        upper_hsv = UPPER_BLUE
    else:
        print(f"  [ERROR] Invalid CARD_SIDE: '{CARD_SIDE}'. Must be 'front' or 'back'.")
        return "N/A"

    # Create a mask of all "good" (correctly colored) pixels
    good_color_mask = cv2.inRange(hsv_img, lower_hsv, upper_hsv)

    # A "defect" is a pixel that is:
    # 1. In the border area (from border_mask)
    # 2. NOT the correct color (by inverting good_color_mask)
    defect_mask = cv2.bitwise_and(border_mask, cv2.bitwise_not(good_color_mask))

    # Save the defect mask for tuning
    cv2.imwrite("defect_mask.png", defect_mask)
    print("  Saved 'defect_mask.png' (White pixels are defects. Use this to tune HSV values.)")

    # Calculate defect percentage
    defect_pixels = cv2.countNonZero(defect_mask)
    defect_percentage = (defect_pixels / total_border_pixels) * 100

    report = f"{defect_percentage:.4f}% ({defect_pixels} defect pixels found)"
    print("  Edge wear analysis complete.")
    return report

def analyze_surface(img_gray, img_color, inner_box):
    """
    Analyzes the inner artwork box for:
    1. Print Lines (Hough Line Transform)
    2. Surface Scuffs (Laplacian Variance)
    """
    print("Step 3: Analyzing Surface (Print Lines & Scuffs)...")
    
    x_i, y_i, w_i, h_i = inner_box

    # Crop the image to *only* the inner artwork box
    artwork_crop_gray = img_gray[y_i:y_i+h_i, x_i:x_i+w_i]
    
    if artwork_crop_gray.size == 0:
        print("  [ERROR] Artwork crop is empty.")
        return "N/A", "N/A"

    # --- Part 1: Print Line Detection ---
    edges = cv2.Canny(artwork_crop_gray, SURFACE_CANNY_THRESHOLDS[0], SURFACE_CANNY_THRESHOLDS[1])
    
    lines = cv2.HoughLinesP(
        edges,
        rho=1,
        theta=np.pi / 180,
        threshold=HOUGH_THRESHOLD,
        minLineLength=HOUGH_MIN_LINE_LENGTH,
        maxLineGap=HOUGH_MAX_LINE_GAP
    )

    line_count = 0
    if lines is not None:
        line_count = len(lines)
        for line in lines:
            x1, y1, x2, y2 = line[0]
            # Draw the detected line on the main visualization image
            # (Remember to add the inner_box offset)
            cv2.line(img_color, (x1 + x_i, y1 + y_i), (x2 + x_i, y2 + y_i), (255, 0, 0), 2)
    
    line_report = f"{line_count} significant lines detected"
    print(f"  {line_report}.")

    # --- Part 2: Surface Scuff/Texture Analysis ---
    # The variance of the Laplacian is a good measure of "texture" or "busyness".
    # A high variance can mean lots of micro-scratches.
    # A low variance means a very smooth, clean surface.
    laplacian_var = cv2.Laplacian(artwork_crop_gray, cv2.CV_64F).var()
    
    scuff_report = f"{laplacian_var:.2f} (Compare this to a known mint card)"
    print(f"  Surface variance: {scuff_report}.")
    print("  Surface analysis complete.")
    
    return line_report, scuff_report

def main():
    """
    Main function to run the full analysis pipeline.
    """
    print(f"Loading image: {IMAGE_FILE}\n")
    try:
        img_color = cv2.imread(IMAGE_FE)
        if img_color is None:
            raise IOError(f"File not found or unable to read: {IMAGE_FILE}")
        
        img_gray = cv2.cvtColor(img_color, cv2.COLOR_BGR2GRAY)
        
        # Create a copy for drawing visualizations
        viz_image = img_color.copy()

    except Exception as e:
        print(f"[FATAL ERROR] Could not load image: {e}")
        print("Please ensure the file path in IMAGE_FILE is correct.")
        return

    # --- Run Analysis Steps ---
    centering_report, outer_box, inner_box = analyze_centering(
        img_gray, viz_image, CENTERİNG_CANNY_THRESHOLDS
    )

    if centering_report is None:
        print("\nFatal error in centering. Aborting further analysis.")
        return

    edge_wear_report = analyze_edge_wear(img_color, outer_box, inner_box)
    
    line_report, scuff_report = analyze_surface(
        img_gray, viz_image, inner_box
    )
    
    # --- Save Visualization ---
    output_image_file = "analysis_visualization.png"
    cv2.imwrite(output_image_file, viz_image)
    print(f"\nSaved visualization to '{output_image_file}'")

    # --- Print Final Report ---
    print("\n--- FINAL CARD ANALYSIS REPORT ---")
    print("====================================")
    print(f"FILE: {IMAGE_FILE}\n")
    print("1. CENTERING:")
    print(centering_report)
    print("\n2. EDGE WEAR (Whitening/Silvering):")
    print(f"  Defect Level: {edge_wear_report}")
    print("\n3. SURFACE ANALYSIS (Artwork Box):")
    print(f"  Print Lines:  {line_report}")
    print(f"  Holo Scuffs:  {scuff_report}")
    print("====================================")

if __name__ == "__main__":
    main()

How to Use This Tool
 * Prerequisites: You must have Python, numpy, and opencv-python installed.
   pip install opencv-python-headless numpy

 * Save the Code: Save the script above as advanced_card_analyzer.py.
 * Get Your Scan: Place your card on a high-resolution flatbed scanner against a plain, dark, contrasting background. A phone photo will not work well.
 * Edit the CONFIG Block:
   * Change IMAGE_FILE to match your scan's filename.
   * Set CARD_SIDE to either 'front' or 'back'. This is essential for the edge wear analysis to work.
 * Run the Script:
   python advanced_card_analyzer.py
